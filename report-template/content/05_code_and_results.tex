\subsection{Vitis HLS C/C++}

\begin{lstlisting}[style=vitisHLS,caption={Illustrative HLS accumulation loop.}]
void energy_accel(const float x[256], float &energy) {
    float acc = 0.0f;
    for (int n = 0; n < 256; ++n) {
        #pragma HLS PIPELINE II=1
        acc += x[n] * x[n];
    }
    energy = acc;
}
\end{lstlisting}
